#include <sstream>
#include <iomanip>

// Dump `size` bytes starting at `address` as a formatted hex string.
// Returns the string; also optionally log via LOG_INFO.
std::string HexDump(uintptr_t address, size_t size, bool logToConsole = true)
{
    std::ostringstream oss;
    // Reserve enough buffer
    oss << "HEX DUMP of 0x" << std::hex << address << " (" << std::dec << size << " bytes)\n";
    oss << "-----------------------------------------------------------\n";

    // Use a fixed width for offset and hex
    std::vector<unsigned char> buffer(size);
    SIZE_T bytesRead = 0;
    if (!ReadProcessMemory(Engine::Get().GetProcessHandle(), (LPCVOID)address,
                           buffer.data(), size, &bytesRead) || bytesRead < size)
    {
        oss << "Failed to read memory at 0x" << std::hex << address << "\n";
        if (logToConsole) LOG_ERROR("%s", oss.str().c_str());
        return oss.str();
    }

    for (size_t i = 0; i < size; i += 16)
    {
        oss << std::setfill('0') << std::setw(8) << std::hex << (address + i) << "  ";

        // Print hex bytes
        for (size_t j = 0; j < 16; ++j)
        {
            if (i + j < size)
            {
                oss << std::setw(2) << static_cast<int>(buffer[i + j]) << " ";
            }
            else
            {
                oss << "   ";
            }
        }

        // Print ASCII representation
        oss << " ";
        for (size_t j = 0; j < 16; ++j)
        {
            if (i + j < size)
            {
                unsigned char c = buffer[i + j];
                if (c >= 32 && c <= 126)
                    oss << c;
                else
                    oss << '.';
            }
        }
        oss << "\n";
    }

    if (logToConsole) LOG_INFO("%s", oss.str().c_str());
    return oss.str();
}
